# H200 数据增强 PI-State 训练命令行

目标：在增强数据集 `hlx` 上训练 QwenPI baseline，并显式启用 state，保证后续 AWAC/Critic/Actor 后训练、rollout LeRobot 诊断数据都能直接接上。

增强数据集路径：

```text
/inspire/qb-ilm2/project/26summer-camp-10/public/inspire_shared/calvin_abc_d/hlx
```

严格约束：

- 本文档跑的是 **QwenPI + state**，不是纯视觉 PI。
- `H200_CALVIN_DATA_NAME=hlx/calvin_task_ABC_D` 必须和 `H200_CALVIN_DATA_MIX=calvin_hlx_h200` 同时使用。
- `calvin_hlx_h200` 在 `examples/calvin/train_files/data_registry/data_config.py` 中必须指向 `("hlx/calvin_task_ABC_D", 1.0, "libero_franka")`。
- `INCLUDE_STATE=true`、`STATE_DIM=8`、`PI_STATE_DIM=8`，与后续 AWAC state 输入保持一致。
- AWAC 前必须跑 `prepare_awac_rewards.py` 写入 `step_reward`、`reward`、`done`；默认不允许缺少 `success` 列时假设成功。

GPU 分配：

| Model | GPUs | Processes | Port |
| --- | --- | --- | --- |
| Qwen3.5-0.8B | `0` | `1` | `35100` |
| Qwen3.5-4B | `1,2` | `2` | `35110` |
| Qwen3.5-9B | `3,4,5,6,7` | `5` | `35120` |

qwen_hidden_size() {
  "${STAR_VLA_PYTHON}" - "$1" <<'PY'
import json
import sys

model_dir = sys.argv[1]
with open(f"{model_dir}/config.json", "r", encoding="utf-8") as f:
    cfg = json.load(f)
text_cfg = cfg.get("text_config") if isinstance(cfg.get("text_config"), dict) else {}
vision_cfg = cfg.get("vision_config") if isinstance(cfg.get("vision_config"), dict) else {}
hidden = cfg.get("hidden_size") or text_cfg.get("hidden_size") or vision_cfg.get("out_hidden_size")
if not isinstance(hidden, int) or hidden <= 0:
    raise SystemExit(f"cannot resolve hidden size from {model_dir}/config.json")
print(hidden)
PY
}

check_hlx_storage() {
  mkdir -p "${BATCH_ROOT}"
  echo "===== storage preflight ====="
  df -h "${PROJECT_ROOT}" "${PROJECT_ROOT}/logs" "${BATCH_ROOT}" 2>/dev/null || true
  df -hi "${PROJECT_ROOT}" "${PROJECT_ROOT}/logs" "${BATCH_ROOT}" 2>/dev/null || true
  quota -s 2>/dev/null || true

  local available_kb
  available_kb="$(df -Pk "${BATCH_ROOT}" | awk 'NR==2 {print $4}')"
  local required_kb=$((MIN_LOG_FREE_GB * 1024 * 1024))
  if [[ -n "${available_kb}" && "${available_kb}" =~ ^[0-9]+$ && "${available_kb}" -lt "${required_kb}" ]]; then
    echo "free space below MIN_LOG_FREE_GB=${MIN_LOG_FREE_GB}GB: available_kb=${available_kb}" >&2
    return 2
  fi

  local probe="${BATCH_ROOT}/.quota_probe_hlx_pi_state_$$"
  if ! dd if=/dev/zero of="${probe}" bs=1M count=64 status=none; then
    rm -f "${probe}" 2>/dev/null || true
    echo "quota probe failed in ${BATCH_ROOT}; free quota before launching." >&2
    return 2
  fi
  rm -f "${probe}"
  sync
  echo "===== storage preflight OK ====="
}

PY
}

launch_three_hlx_pi_state() {
  H200_HLX_PIDS=()
  H200_HLX_NAMES=()
  H200_HLX_LOG_DIRS=()

  check_hlx_dataset
  check_hlx_storage

  launch_hlx_qwen_size qwen35_0p8b "${QWEN35_0P8B}" 0 1 35100 "${PER_DEVICE_BATCH_SIZE_0P8B}"
  launch_hlx_qwen_size qwen35_4b "${QWEN35_4B}" 1,2 2 35110 "${PER_DEVICE_BATCH_SIZE_4B}"
  launch_hlx_qwen_size qwen35_9b "${QWEN35_9B}" 3,4,5,6,7 5 35120 "${PER_DEVICE_BATCH_SIZE_9B}"

  local status=0
  local idx
  for idx in "${!H200_HLX_PIDS[@]}"; do
    local pid="${H200_HLX_PIDS[$idx]}"
    local name="${H200_HLX_NAMES[$idx]}"
    local log_dir="${H200_HLX_LOG_DIRS[$idx]}"
    if ! wait "${pid}"; then
      echo "FAILED model=${name} pid=${pid} log_dir=${log_dir}" >&2
      tail -n 160 "${log_dir}/terminal/launcher.log" >&2 || true
      status=1
    else
      echo "DONE model=${name} pid=${pid} log_dir=${log_dir}"
    fi
  done
  return "${status}"
}

## 6. 常见失败判断

| 现象 | 判断 | 处理 |
| --- | --- | --- |
| registry 检查显示 `actual=None` | 代码没有包含 `calvin_hlx_h200` | 先更新代码并 `git pull` |
| 训练日志里使用旧根目录 `calvin_abc_d_h200` 或旧数据 `calvin_task_ABC_D` | mixture 或环境变量错了 | 停止训练，确认 `H200_CALVIN_DATA_NAME=hlx/calvin_task_ABC_D` 且 `H200_CALVIN_DATA_MIX=calvin_hlx_h200` |
| `features.state.shape` 不是 `[8]` | 数据不是当前 CALVIN state 格式 | 不要强跑，先修数据格式 |
| `prepare_awac_rewards.py --dry_run` 报缺少 `success` | 后训练 reward 无可靠成功标签 | 补 success，或明确确认全成功后再使用 `--assume_success_if_missing` |
| policy eval 成功启动但无 rollout parquet | 没打开 rollout 写盘 | 确认 `WRITE_ROLLOUT_LEROBOT=1` |
| PI-State eval 表现异常 | state 没发给 policy 或 state 维度不一致 | 确认 `SEND_STATE_TO_POLICY=1`、`STATE_DIM=8` |
