# H200 CALVIN ABC->D 批量测评命令行

目标：在 H200 离线服务器上，对各 baseline 的三种 Qwen3.5 权重跑完整 CALVIN ABC->D 测评，只输出 `results.json` 和汇总表，不输出 mp4。最终指标只看 Task 1~5 成功率与平均链长。

固定要求：

- 测试数据集：`/inspire/qb-ilm2/project/26summer-camp-10/public/inspire_shared/calvin_d_d`
- 输出根目录：`/inspire/qb-ilm2/project/26summer-camp-10/26220216/starVLA_Project/results/model_eval`
- 不写 mp4：`DEBUG_MP4=0`、`REQUIRE_MP4=0`、`WRITE_ROLLOUT_* = 0`
- 离线运行：不依赖网络，不访问 HuggingFace/W&B
- 默认每个模型跑 `1000` 条 eval sequences 作为分母
- 默认一个模型一次占用 8 张 H200，把 1000 条序列切成 8 个不重叠 shard 并行跑；模型之间串行，避免一次加载太多模型导致服务器卡死

注意：`eval_calvin.py` 的 `sequence_index_offset` 只影响编号，不负责切片。本文档用生成 shard 版 `eval_sequences.json` 的方式并行，避免 8 个 worker 重复测同一批序列。

```bash
latest_step_for_job() {
  local batch_root=$1
  local job=$2
  "${STAR_VLA_PYTHON}" - "${batch_root}" "${job}" <<'PY'
import re
import sys
from pathlib import Path

root = Path(sys.argv[1]) / sys.argv[2]
pattern = re.compile(r"steps_(\d+)_pytorch_model\.pt$")
candidates = []
for path in root.rglob("steps_*_pytorch_model.pt"):
    match = pattern.search(path.name)
    if match:
        candidates.append((int(match.group(1)), path))
if not candidates:
    raise SystemExit(f"no checkpoint under {root}")
step, path = max(candidates, key=lambda item: item[0])
print(step)
PY
}

write_shard_jsons() {
  local shard_dir=$1
  "${STAR_VLA_PYTHON}" - "${EVAL_SEQUENCES_PATH}" "${shard_dir}" "${TOTAL_SEQUENCES}" "${SHARDS}" <<'PY'
import json
import sys
from pathlib import Path

src = Path(sys.argv[1])
out = Path(sys.argv[2])
total = int(sys.argv[3])
shards = int(sys.argv[4])
if total < 1 or shards < 1:
    raise SystemExit("TOTAL_SEQUENCES and SHARDS must be positive")
with src.open("r", encoding="utf-8") as f:
    sequences = json.load(f)
if total > len(sequences):
    raise SystemExit(f"TOTAL_SEQUENCES={total} exceeds available eval sequences={len(sequences)}")
out.mkdir(parents=True, exist_ok=True)
manifest = []
base = total // shards
rem = total % shards
offset = 0
for shard in range(shards):
    count = base + (1 if shard < rem else 0)
    shard_sequences = sequences[offset:offset + count]
    path = out / f"shard_{shard:02d}.json"
    with path.open("w", encoding="utf-8") as f:
        json.dump(shard_sequences, f)
    manifest.append({"shard": shard, "offset": offset, "count": count, "path": str(path)})
    offset += count
with (out / "shards.json").open("w", encoding="utf-8") as f:
    json.dump(manifest, f, indent=2)
print(f"wrote_shards={len(manifest)} total={total} dir={out}")
PY
}

aggregate_results() {
  local run_root=$1
  "${STAR_VLA_PYTHON}" - "${run_root}" "${MODEL_EVAL_MANIFEST}" "${TOTAL_SEQUENCES}" <<'PY'
import csv
import json
import re
import sys
from pathlib import Path
from typing import Any

run_root = Path(sys.argv[1])
manifest_path = Path(sys.argv[2])
expected_total = int(sys.argv[3])

RAW_KEYS = {"results", "successes", "sequence_results", "successful_tasks", "success_counter"}

def numeric(value: Any):
    if isinstance(value, bool):
        return int(value)
    if isinstance(value, (int, float)):
        return int(value)
    if isinstance(value, str):
        m = re.search(r"-?\d+", value)
        return int(m.group(0)) if m else None
    return None

def list_counts(obj: Any):
    if isinstance(obj, list):
        out = []
        for item in obj:
            if isinstance(item, dict):
                val = None
                for key in ("success_counter", "successful_tasks", "num_success", "result", "success"):
                    if key in item:
                        val = numeric(item[key])
                        break
            else:
                val = numeric(item)
            if val is None or val < 0 or val > 5:
                return None
            out.append(val)
        return out if out else None
    return None

def walk(obj: Any):
    if isinstance(obj, dict):
        yield obj
        for value in obj.values():
            yield from walk(value)
    elif isinstance(obj, list):
        for value in obj:
            yield from walk(value)

def extract_counts(path: Path):
    data = json.loads(path.read_text(encoding="utf-8-sig"))
    direct = list_counts(data)
    if direct and len(direct) > 5:
        return direct
    for mapping in walk(data):
        for key, value in mapping.items():
            if str(key).lower() in RAW_KEYS or "result" in str(key).lower():
                vals = list_counts(value)
                if vals and len(vals) > 5:
                    return vals
    raise ValueError(f"cannot extract per-sequence success counts from {path}")

rows = []
with manifest_path.open("r", encoding="utf-8-sig", newline="") as f:
    reader = csv.DictReader(f, delimiter="\t")
    wanted = [dict(row) for row in reader]

for item in wanted:
    run_name = item["run_name"]
    job = item["job"]
    safe_job = job.replace("/", "__")
    model_root = run_root / run_name / safe_job
    counts = []
    result_files = []
    failed = []
    for summary in sorted(model_root.glob("shard_*/summary.tsv")):
        with summary.open("r", encoding="utf-8-sig", newline="") as f:
            reader = csv.DictReader(f, delimiter="\t")
            for row in reader:
                if row.get("status") != "OK":
                    failed.append(f"{summary}: status={row.get('status')}")
                    continue
                result = Path(row["results_json"])
                result_files.append(str(result))
                counts.extend(extract_counts(result))
    status = "OK"
    if failed:
        status = "FAILED"
    if len(counts) != expected_total:
        status = f"BAD_COUNT_{len(counts)}"
    denom = len(counts) if counts else 1
    chain = [sum(value >= k for value in counts) / denom for k in range(1, 6)] if counts else [0.0] * 5
    avg = sum(counts) / denom if counts else 0.0
    rows.append({
        "run_name": run_name,
        "job": job,
        "num_sequences": str(len(counts)),
        "task1_sr": f"{chain[0]:.6f}",
        "task2_sr": f"{chain[1]:.6f}",
        "task3_sr": f"{chain[2]:.6f}",
        "task4_sr": f"{chain[3]:.6f}",
        "task5_sr": f"{chain[4]:.6f}",
        "avg_chain_len": f"{avg:.6f}",
        "status": status,
        "result_files": ";".join(result_files),
    })

tsv = run_root / "final_summary.tsv"
md = run_root / "final_summary.md"
fields = ["run_name", "job", "num_sequences", "task1_sr", "task2_sr", "task3_sr", "task4_sr", "task5_sr", "avg_chain_len", "status", "result_files"]
with tsv.open("w", encoding="utf-8", newline="") as f:
    writer = csv.DictWriter(f, fieldnames=fields, delimiter="\t")
    writer.writeheader()
    writer.writerows(rows)
with md.open("w", encoding="utf-8") as f:
    f.write("# CALVIN ABC->D Eval Summary\n\n")
    f.write(f"- expected_num_sequences: `{expected_total}`\n")
    f.write(f"- run_root: `{run_root}`\n\n")
    f.write("| run | job | N | Task1 | Task2 | Task3 | Task4 | Task5 | AvgLen | status |\n")
    f.write("| --- | --- | ---: | ---: | ---: | ---: | ---: | ---: | ---: | --- |\n")
    for row in rows:
        f.write(
            f"| {row['run_name']} | {row['job']} | {row['num_sequences']} | "
            f"{float(row['task1_sr'])*100:.2f}% | {float(row['task2_sr'])*100:.2f}% | "
            f"{float(row['task3_sr'])*100:.2f}% | {float(row['task4_sr'])*100:.2f}% | "
            f"{float(row['task5_sr'])*100:.2f}% | {float(row['avg_chain_len']):.3f} | {row['status']} |\n"
        )
print(f"WROTE {tsv}")
print(f"WROTE {md}")
bad = [row for row in rows if row["status"] != "OK"]
if bad:
    for row in bad:
        print(f"BAD {row['run_name']} {row['job']} status={row['status']} N={row['num_sequences']}", file=sys.stderr)
    raise SystemExit(3)
PY
}

run_one_model() {
  local run_name=$1
  local batch_root=$2
  local send_state=$3
  local job=$4
  local min_step=$5

  local step
  step="$(latest_step_for_job "${batch_root}" "${job}")"
  if (( step < min_step )); then
    echo "checkpoint step too small: run=${run_name} job=${job} step=${step} min_step=${min_step}" >&2
    return 2
  fi

  local safe_job="${job//\//__}"
  local run_root="${MODEL_EVAL_ROOT}/${RUN_ID}"
  local model_root="${run_root}/${run_name}/${safe_job}"
  local shard_dir="${model_root}/sequence_shards"
  local terminal_dir="${model_root}/terminal"
  mkdir -p "${terminal_dir}"
  write_shard_jsons "${shard_dir}"

  local pids=()
  local ports=()
  local logs=()
  local shard
  for (( shard=0; shard<SHARDS; shard++ )); do
    local gpu=$(( shard % 8 ))
    local port=$(( BASE_PORT + shard ))
    local shard_json="${shard_dir}/shard_$(printf '%02d' "${shard}").json"
    local shard_root="${model_root}/shard_$(printf '%02d' "${shard}")"
    local log="${terminal_dir}/shard_$(printf '%02d' "${shard}").log"
    mkdir -p "${shard_root}"
    cleanup_port "${port}"
    (
      set -euo pipefail
      export PROJECT_ROOT CONDA_ROOT STAR_VLA_PYTHON CALVIN_PYTHON
      export H200_CALVIN_EVAL_DATASET_PATH
      export EVAL_SEQUENCES_PATH="${shard_json}"
      export EVAL_BATCH_ROOT="${batch_root}"
      export EVAL_OUTPUT_ROOT="${shard_root}"
      export EXPECTED_JOBS="${job}"
      export EVAL_GPU="${gpu}"
      export EVAL_PORT="${port}"
      export NUM_SEQUENCES=0
      export EVAL_SEQUENCE_INDEX_OFFSET=0
      export FAIL_FAST=1
      export REQUIRE_MP4=0
      export DEBUG_MP4=0
      export SEND_STATE_TO_POLICY="${send_state}"
      export WRITE_ROLLOUT_LEROBOT=0
      export WRITE_ROLLOUT_VIDEOS=0
      export UNNORM_KEY=franka
      export POLICY_SERVER_START_TIMEOUT=900
      export PYOPENGL_PLATFORM MUJOCO_GL GIT_PYTHON_REFRESH CALVIN_ALLOW_OFFLINE_GIT_METADATA CALVIN_FORCE_NO_EGL
      export NO_ALBUMENTATIONS_UPDATE TOKENIZERS_PARALLELISM HF_HUB_OFFLINE TRANSFORMERS_OFFLINE WANDB_MODE
      timeout --kill-after=120s "${WORKER_TIMEOUT}" bash examples/calvin/eval_files/eval_h200_qwen_mix_latest.sh
      cleanup_port "${port}"
    ) > "${log}" 2>&1 &
    pids+=("$!")
    ports+=("${port}")
    logs+=("${log}")
    echo "LAUNCHED run=${run_name} job=${job} shard=${shard}/${SHARDS} gpu=${gpu} port=${port} log=${log}"
  done

  local status=0
  local done_count=0
  while (( done_count < ${#pids[@]} )); do
    done_count=0
    local i
    for i in "${!pids[@]}"; do
      if ! kill -0 "${pids[$i]}" 2>/dev/null; then
        done_count=$((done_count + 1))
        continue
      fi
      local now
      local mtime
      now="$(date +%s)"
      mtime="$(stat -c %Y "${logs[$i]}" 2>/dev/null || echo "${now}")"
      if (( now - mtime > STALL_TIMEOUT )); then
        echo "STALL detected: pid=${pids[$i]} port=${ports[$i]} log=${logs[$i]}" >&2
        kill "${pids[$i]}" 2>/dev/null || true
        sleep 5
        kill -9 "${pids[$i]}" 2>/dev/null || true
        cleanup_port "${ports[$i]}"
        status=1
      fi
    done
    sleep 60
  done

  local i
  for i in "${!pids[@]}"; do
    if wait "${pids[$i]}"; then
      echo "DONE run=${run_name} job=${job} shard=${i} log=${logs[$i]}"
    else
      echo "FAILED run=${run_name} job=${job} shard=${i} log=${logs[$i]}" >&2
      tail -n 160 "${logs[$i]}" >&2 || true
      status=1
    fi
    cleanup_port "${ports[$i]}"
  done
  return "${status}"
}

export RUN_ID="abc_d_${TOTAL_SEQUENCES}seq_$(date +%Y%m%d_%H%M%S)"
export RUN_ROOT="${MODEL_EVAL_ROOT}/${RUN_ID}"
mkdir -p "${RUN_ROOT}"
cp "${MODEL_EVAL_MANIFEST}" "${RUN_ROOT}/model_manifest.tsv"

status=0
tail -n +2 "${MODEL_EVAL_MANIFEST}" | while IFS=$'\t' read -r run_name batch_root send_state job min_step; do
  [[ -n "${run_name}" ]] || continue
  if ! run_one_model "${run_name}" "${batch_root}" "${send_state}" "${job}" "${min_step}"; then
    status=1
    echo "MODEL_FAILED run=${run_name} job=${job}" >&2
  fi
done

aggregate_results "${RUN_ROOT}" || status=1
echo "RUN_ROOT=${RUN_ROOT}"
echo "FINAL_MD=${RUN_ROOT}/final_summary.md"
echo "FINAL_TSV=${RUN_ROOT}/final_summary.tsv"
exit "${status}"
BASH
```

Smoke 通过标准：
